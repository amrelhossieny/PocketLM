## 1. App Overview 🎯

### What is PocketLM?
PocketLM is a **free, open-source, privacy-first** mobile app that lets users download and run Large Language Models (LLMs) directly on their phone - **completely offline** with no subscription needed.

### Core Values
```
🔒 Privacy First    → Everything runs on device
💰 Free Forever     → No subscription, no hidden fees
🌐 Offline First    → No internet needed after download
🎯 User Friendly    → Anyone can use it, not just developers
⚡ Performance      → Optimized for mobile hardware
```

### Target Audience
| User Type | Why They'd Use PocketLM |
|---|---|
| **Privacy Conscious** | No data sent to cloud |
| **Developers** | Test models locally |
| **Students** | Free AI assistant |
| **Travelers** | Works without internet |
| **Enthusiasts** | Explore open source AI |

### 4.2 LLM Calculator (Compatibility Checker)

**Goal:** Calculate if a model can run on device based on specs

**Reference:** [LLM-Calc by RayFernando1337](https://github.com/RayFernando1337/LLM-Calc)

```dart
// llm_calculator.dart

class LLMCalculator {
  
  /// Main compatibility check
  static ModelCompatibility checkCompatibility({
    required PhoneSpecs specs,
    required HFModel model,
    int contextWindowSize = 2048,
  }) {
    final usableRAM = _calculateUsableRAM(
      availableRAM: specs.availableRAM,
      contextWindowSize: contextWindowSize,
    );
    
    final maxParams = _calculateMaxParams(
      usableRAM: usableRAM,
      quantizationBits: model.quantizationBits,
    );
    
    final status = _getCompatibilityStatus(
      modelParams: model.parameters,
      maxParams: maxParams,
    );
    
    return ModelCompatibility(
      canRun: status != CompatibilityStatus.tooLarge,
      status: status,
      maxParameters: maxParams,
      usableRAM: usableRAM,
      recommendedContextSize: contextWindowSize,
    );
  }

  /// Calculate usable RAM after OS and context window overhead
  static double _calculateUsableRAM({
    required double availableRAM,
    required int contextWindowSize,
  }) {
    // Convert to bytes
    final ramBytes = availableRAM * 1024 * 1024 * 1024;
    
    // Context window memory (bytes)
    // Formula from LLM-Calc
    final contextMemory = contextWindowSize * 2 * 2;
    
    // Usable RAM in GB
    return (ramBytes - contextMemory) / (1024 * 1024 * 1024);
  }

  /// Calculate max parameters device can handle
  static double _calculateMaxParams({
    required double usableRAM,
    required int quantizationBits,
  }) {
    // Convert bits to bytes per parameter
    final bytesPerParam = quantizationBits / 8;
    
    // Convert usable RAM to bytes
    final usableBytes = usableRAM * 1024 * 1024 * 1024;
    
    // Max params in billions
    return (usableBytes / bytesPerParam) / 1e9;
  }

  /// Get compatibility status
  static CompatibilityStatus _getCompatibilityStatus({
    required double modelParams,
    required double maxParams,
  }) {
    final ratio = modelParams / maxParams;
    
    if (ratio <= 0.7) return CompatibilityStatus.great;
    if (ratio <= 0.9) return CompatibilityStatus.good;
    if (ratio <= 1.0) return CompatibilityStatus.tight;
    return CompatibilityStatus.tooLarge;
  }
}

enum CompatibilityStatus { great, good, tight, tooLarge }

class ModelCompatibility {
  final bool canRun;
  final CompatibilityStatus status;
  final double maxParameters;
  final double usableRAM;
  final int recommendedContextSize;
}
```

**Compatibility UI:**
```
🟢 Great    → Model uses less than 70% of available RAM
🟡 Good     → Model uses 70-90% of available RAM  
🟠 Tight    → Model uses 90-100% of available RAM
🔴 Too Big  → Model exceeds available RAM
```

### 4.3 HuggingFace Explore Page

**Goal:** Browse and discover AI models from HuggingFace

**API Reference:** [HuggingFace API Docs](https://huggingface.co/docs/api-inference)

```dart
// huggingface_service.dart

class HuggingFaceService {
  final Dio _dio;
  static const String baseUrl = 'https://huggingface.co/api';
  
  /// Get trending GGUF models
  Future<List<HFModel>> getTrendingModels({
    int limit = 20,
    int offset = 0,
  }) async {
    final response = await _dio.get(
      '$baseUrl/models',
      queryParameters: {
        'library': 'gguf',          // Only GGUF models
        'sort': 'trending',
        'limit': limit,
        'offset': offset,
        'full': true,               // Get full model info
      },
    );
    
    return (response.data as List)
        .map((m) => HFModel.fromJson(m))
        .toList();
  }
  
  /// Search models
  Future<List<HFModel>> searchModels({
    required String query,
    int limit = 20,
  }) async {
    final response = await _dio.get(
      '$baseUrl/models',
      queryParameters: {
        'search': query,
        'library': 'gguf',
        'limit': limit,
        'full': true,
      },
    );
    
    return (response.data as List)
        .map((m) => HFModel.fromJson(m))
        .toList();
  }

  /// Get model details
  Future<HFModel> getModelDetails(String modelId) async {
    final response = await _dio.get('$baseUrl/models/$modelId');
    return HFModel.fromJson(response.data);
  }
  
  /// Get model files (to find GGUF files)
  Future<List<ModelFile>> getModelFiles(String modelId) async {
    final response = await _dio.get(
      '$baseUrl/models/$modelId',
      queryParameters: {'blobs': true},
    );
    return (response.data['siblings'] as List)
        .map((f) => ModelFile.fromJson(f))
        .where((f) => f.filename.endsWith('.gguf'))
        .toList();
  }
}
```

### 4.4 Download & Manage Models

**Goal:** Download models from HuggingFace and manage local storage

```dart
// download_service.dart

class DownloadService {
  
  /// Start model download
  Future<void> downloadModel({
    required HFModel model,
    required ModelFile file,
    required Function(double progress) onProgress,
  }) async {
    final savePath = await _getModelSavePath(file.filename);
    
    await _dio.download(
      file.downloadUrl,
      savePath,
      onReceiveProgress: (received, total) {
        if (total != -1) {
          onProgress(received / total);
        }
      },
    );
    
    // Save model info to local DB
    await _saveModelToDb(model, file, savePath);
  }
  
  /// Get all downloaded models
  Future<List<DownloadedModel>> getDownloadedModels() async {
    return await _modelBox.values.toList();
  }
  
  /// Delete a model
  Future<void> deleteModel(String modelId) async {
    final model = await _modelBox.get(modelId);
    if (model != null) {
      await File(model.localPath).delete();
      await _modelBox.delete(modelId);
    }
  }
  
  /// Get models directory path
  Future<String> _getModelSavePath(String filename) async {
    final dir = await getApplicationDocumentsDirectory();
    final modelsDir = Directory('${dir.path}/models');
    if (!await modelsDir.exists()) {
      await modelsDir.create(recursive: true);
    }
    return '${modelsDir.path}/$filename';
  }
}
```

### 4.5 Basic Chat UI

**Goal:** Chat with downloaded models using llama.cpp

**Reference:** [llama_cpp_dart](https://github.com/netdur/llama_cpp_dart)

```dart
// llama_service.dart

class LlamaService {
  LlamaContext? _context;
  
  /// Load a model
  Future<void> loadModel(DownloadedModel model) async {
    final params = ModelParams(
      nGpuLayers: 0,       // CPU only for now
    );
    
    _context = await LlamaContext.create(
      modelPath: model.localPath,
      params: params,
    );
  }
  
  /// Generate response (streaming)
  Stream<String> chat({
    required String message,
    required List<ChatMessage> history,
  }) async* {
    if (_context == null) throw Exception('No model loaded');
    
    final prompt = _buildPrompt(message, history);
    
    final params = SamplingParams(
      temperature: 0.7,
      topP: 0.9,
      maxTokens: 512,
    );
    
    await for (final token in _context!.generate(prompt, params)) {
      yield token;
    }
  }
  
  /// Build prompt from history
  String _buildPrompt(
    String message,
    List<ChatMessage> history,
  ) {
    final buffer = StringBuffer();
    
    // System prompt
    buffer.writeln('<|system|>');
    buffer.writeln('You are a helpful AI assistant.');
    buffer.writeln('<|end|>');
    
    // Chat history
    for (final msg in history) {
      if (msg.isUser) {
        buffer.writeln('<|user|>');
        buffer.writeln(msg.content);
        buffer.writeln('<|end|>');
      } else {
        buffer.writeln('<|assistant|>');
        buffer.writeln(msg.content);
        buffer.writeln('<|end|>');
      }
    }
    
    // Current message
    buffer.writeln('<|user|>');
    buffer.writeln(message);
    buffer.writeln('<|end|>');
    buffer.writeln('<|assistant|>');
    
    return buffer.toString();
  }
  
  /// Unload model from memory
  Future<void> unloadModel() async {
    await _context?.dispose();
    _context = null;
  }
}
```

### 5.3 Download Resume Support

```dart
class DownloadService {
  
  /// Download with resume support
  Future<void> downloadWithResume({
    required String url,
    required String savePath,
    required Function(double) onProgress,
  }) async {
    final file = File(savePath);
    int downloadedBytes = 0;
    
    // Check if partial download exists
    if (await file.exists()) {
      downloadedBytes = await file.length();
    }
    
    await _dio.download(
      url,
      savePath,
      options: Options(
        headers: {
          // Resume from where we left off
          'Range': 'bytes=$downloadedBytes-',
        },
        responseType: ResponseType.stream,
      ),
      onReceiveProgress: (received, total) {
        final progress = (downloadedBytes + received) / 
                        (downloadedBytes + total);
        onProgress(progress);
      },
    );
  }
}
```

```dart
// ollama_service.dart

class OllamaService {
  final Dio _dio;
  String? _baseUrl;        // User's Ollama server URL
  String? _apiKey;         // Optional API key
  
  /// Connect to Ollama instance
  Future<bool> connect({
    required String serverUrl,
    String? apiKey,
  }) async {
    try {
      _baseUrl = serverUrl;
      _apiKey = apiKey;
      
      // Test connection
      final response = await _dio.get('$serverUrl/api/tags');
      return response.statusCode == 200;
    } catch (e) {
      return false;
    }
  }
  
  /// Get available models on Ollama server
  Future<List<OllamaModel>> getModels() async {
    final response = await _dio.get('$_baseUrl/api/tags');
    return (response.data['models'] as List)
        .map((m) => OllamaModel.fromJson(m))
        .toList();
  }
  
  /// Chat with Ollama model
  Stream<String> chat({
    required String model,
    required String message,
    required List<ChatMessage> history,
  }) async* {
    final response = await _dio.post(
      '$_baseUrl/api/chat',
      data: {
        'model': model,
        'messages': [
          ...history.map((m) => {
            'role': m.isUser ? 'user' : 'assistant',
            'content': m.content,
          }),
          {'role': 'user', 'content': message},
        ],
        'stream': true,
      },
      options: Options(responseType: ResponseType.stream),
    );
    
    await for (final chunk in response.data.stream) {
      final data = jsonDecode(utf8.decode(chunk));
      if (data['message'] != null) {
        yield data['message']['content'];
      }
    }
  }
}
```

### 6.2 Cloud Fallback Logic

```dart
// chat_provider.dart

class ChatProvider extends ChangeNotifier {
  
  Future<void> sendMessage(String message) async {
    // Try local model first
    if (_localModelLoaded) {
      yield* _llamaService.chat(
        message: message,
        history: _history,
      );
    }
    // Fallback to Ollama if model too big
    else if (_ollamaService.isConnected) {
      yield* _ollamaService.chat(
        model: _selectedOllamaModel,
        message: message,
        history: _history,
      );
    }
    else {
      throw Exception('No model available!');
    }
  }
}
```
